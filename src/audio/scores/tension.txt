# Tension — the creeping one, for a stage that has stopped being friendly.
# A natural minor, eight bars, slow enough that the rests do most of the work.
#
# The sketch this came from had a square wave pulsing straight eighths all the way
# through, which is the opposite of creeping: a steady pulse is reassuring. What
# unsettles is space, so the loop is mostly space and the motif only half fills it.

tempo 76
loop on

# Am Am F G | Am Dm Em Am. No g#: the dominant here is Em, not E, which leaves the
# whole thing modal and refuses the resolution a major V would promise.
track pad
  voice sine
  gain 0.08
  octave 4
  env 0.5 0.6 0.7 0.9
  | [a c5 e5]/1 |
  | [a c5 e5]/1 |
  | [f a c5]/1  |
  | [g b d5]/1  |
  | [a c5 e5]/1 |
  | [d f a]/1   |
  | [e g b]/1   |
  | [a c5 e5]/1 |

# The floor of the piece. Two bars of A before it moves at all.
track drone
  voice triangle
  gain 0.16
  octave 2
  env 1.0 1.2 0.85 1.5
  | a/1 ~/1 |
  | f/1     |
  | g/1     |
  | a/1     |
  | d/1     |
  | e/1     |
  | a/1     |

# Four notes, said twice, with a bar of nothing after each — the silence is the point.
track motif
  voice sine
  gain 0.11
  octave 5
  env 0.02 0.25 0.4 0.4
  | a/4 -/4 c6/8 b/8 a/4 |
  | -/1                  |
  | f/4 -/4 a/8  g/8 f/4 |
  | -/1                  |
  | a/4 -/4 c6/8 b/8 a/4 |
  | -/2 d6/4 c6/4        |
  | b/2 -/2              |
  | a/1                  |

# Enters only in the second half, and climbs — the one thing in the loop that goes
# somewhere, so the repeat feels like it is tightening rather than just repeating.
track high
  voice sine
  gain 0.05
  octave 6
  env 0.4 0.7 0.5 0.9
  | -/1     |
  | -/1     |
  | -/1     |
  | -/1     |
  | -/2 e/2 |
  | -/2 f/2 |
  | -/2 g/2 |
  | -/1     |

# Something shifting out of sight, never on a beat you could tap to.
track ticks
  voice noise
  gain 0.04
  env 0.001 0.05 0.0 0.06
  | -/1                       |
  | -/2 x/16 -/16 -/8 -/4     |
  | -/1                       |
  | -/4 x/16 -/16 -/8 -/2     |
  | -/1                       |
  | -/2 x/16 -/16 x/16 -/16 -/4 |
  | -/1                       |
  | -/2 x/16 -/16 -/8 -/4     |
