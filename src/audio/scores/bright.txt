# Bright — the cheerful one, for the early stages. C major, eight bars.
#
# Two four-bar phrases: bars 1-4 ask and leave off on G, bars 5-8 answer and land
# home on C. Eight bars rather than four because a four-bar loop at this tempo comes
# round every seven seconds, which is long enough to notice and short enough to grate.

tempo 124
loop on

# C  Am  F  G | C  Am  F G  C — the whole progression, one column you can read down.
# The last bar of each phrase is where they differ: bar 4 hangs on G, bar 7 walks
# F to G so bar 8 can close.
track pad
  voice sine
  gain 0.09
  octave 4
  env 0.25 0.35 0.7 0.5
  | [c e g]/1     |
  | [a c5 e5]/1   |
  | [f a c5]/1    |
  | [g b d5]/1    |
  | [c e g]/1     |
  | [a c5 e5]/1   |
  | [f a c5]/2 [g b d5]/2 |
  | [c e g]/1     |

# Root and fifth, on the beat and off it — the hop the piece walks on.
track bass
  voice triangle
  gain 0.22
  octave 2
  env 0.01 0.07 0.5 0.12
  | c/8 -/8 c/8 -/8 g/8 -/8 c/8 -/8 |
  | a/8 -/8 a/8 -/8 e/8 -/8 a/8 -/8 |
  | f/8 -/8 f/8 -/8 c/8 -/8 f/8 -/8 |
  | g/8 -/8 g/8 -/8 d/8 -/8 g/8 -/8 |
  | c/8 -/8 c/8 -/8 g/8 -/8 c/8 -/8 |
  | a/8 -/8 a/8 -/8 e/8 -/8 a/8 -/8 |
  | f/8 -/8 f/8 -/8 g/8 -/8 g/8 -/8 |
  | c/4 g/8 -/8 c/4 -/4             |

# The tune. Bars 1 and 3 rise and stop for breath; bars 2 and 4 answer them falling.
# The second phrase is the same shape reaching a third higher.
track lead
  voice square
  gain 0.15
  octave 5
  env 0.01 0.10 0.4 0.16
  | e/4  g/4   c6/4 -/4  |
  | a/4  c6/8  b/8  a/2  |
  | f/4  a/4   c6/4 -/4  |
  | g/4  b/8   a/8  g/2  |
  | e/4  g/4   c6/4 e6/4 |
  | d6/8 c6/8  b/4  a/2  |
  | f/4  g/4   a/4  b/4  |
  | c6/2 ~/2             |

# Silent for the whole first phrase, so the second one arrives with something new in
# it. A sine up here glints; a square would shriek.
track sparkle
  voice sine
  gain 0.05
  octave 6
  env 0.005 0.12 0.0 0.2
  | -/1            |
  | -/1            |
  | -/1            |
  | -/1            |
  | -/4 e/8 g/8 -/2 |
  | -/4 c/8 e/8 -/2 |
  | -/4 a/8 c/8 -/2 |
  | -/4 g/8 e/8 -/2 |

# One hat a beat, and a fill in the last bar to hand the loop back to the top.
track hats
  voice noise
  gain 0.05
  env 0.001 0.03 0.0 0.03
  | x/8 -/8 x/8 -/8 x/8 -/8 x/8 -/8       |
  | x/8 -/8 x/8 -/8 x/8 -/8 x/8 -/8       |
  | x/8 -/8 x/8 -/8 x/8 -/8 x/8 -/8       |
  | x/8 -/8 x/8 -/8 x/8 -/8 x/8 -/8       |

  | x/8 -/8 x/8 -/8 x/8 -/8 x/8 -/8       |
  | x/8 -/8 x/8 -/8 x/8 -/8 x/8 -/8       |
  | x/8 -/8 x/8 -/8 x/8 x/16 x/16 x/8 x/8 |
