# Teleport — a quick rising blip and a shimmer over it, so a warp reads as going
# somewhere rather than picking something up.

tempo 210
loop off

# The trip: four notes up, fast enough to hear as one gesture.
track blip
  voice triangle
  gain 0.20
  octave 5
  env 0.005 0.04 0.25 0.10
  | d/16 a/16 d6/16 a6/16 |

# A thin sine above it, arriving a beat later — the far end answering.
track shimmer
  voice sine
  gain 0.10
  octave 6
  env 0.01 0.12 0.20 0.22
  | -/8 f#/8 |
